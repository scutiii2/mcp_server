# Chat drag and drop (phase 5) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A chat row can be dragged onto a folder, the Pinned section or the Chats section of the Ember sidebar to move or pin it, with the same effect as the "..." menu.

**Architecture:** Two pure helpers decide what a drop does (`dropOps`) and which zones to show while dragging (`buildLayout(..., dragging)`). `ChatRow` becomes draggable and `ChatSection` becomes a drop zone; `ConversationSidebar` owns the drag state and turns a drop into its existing `move` and `pin` events, so `ChatView` and the stores do not change. HTML5 drag and drop (no library).

**Tech Stack:** Vue 3 (script setup), TypeScript (`erasableSyntaxOnly`), Vitest + jsdom + @vue/test-utils, Playwright.

**Spec:** `docs/superpowers/specs/2026-10-05-chat-folders-pins-design.md`, section "Phase 5 design (approved 2026-10-06): drag and drop".

## Global Constraints

- Drop on a folder (its header or its rows, also when collapsed): the chat moves into that folder and is unpinned if it was pinned. Drop on Pinned: the chat is pinned and keeps its folder. Drop on Chats: the chat leaves its folder and is unpinned.
- While a chat is being dragged, an empty Pinned section and an empty Chats section are shown as drop zones with a short hint, but only when the list is grouped (there are folders or pins). The flat list (no folders, no pins) has nothing to drop on and does not change.
- A section that would change nothing for the dragged chat does not accept the drop and gets no highlight.
- A chat cannot be dragged when it is answering (the sidebar's `answering(c)`), in select mode, while it is being renamed, on touch screens (`window.matchMedia("(hover: none)").matches`), or in search results.
- The sidebar emits only its existing `move(id, folderId | null)` and `pin(id, pinned)` events for a drop, `move` before `pin`. `ChatView.vue` and the stores are not changed.
- The drag state is set in a `setTimeout(0)` after `dragstart`: Chrome cancels a drag whose source DOM changes inside the `dragstart` handler, and showing the drop zones changes the DOM.
- Theme tokens only; no hardcoded colors. tsconfig `erasableSyntaxOnly`.
- Verify from `ember_web/`: `npx vue-tsc -b --noEmit` silent; `npm test` passes (1247 tests before this work); `npx vite build` succeeds (then delete `dist/`); `npm run test:e2e` passes (only if the Playwright browser is installed; otherwise report NOT RUN). No manual dev server or browser.

```ts
import type { Conversation } from "../api/types";

/** Where a dragged chat was dropped. */
export type DropTarget = { kind: "pinned" } | { kind: "folder"; folderId: number } | { kind: "unfiled" };

/** One change a drop makes; `move` always comes before `pin`. */
export type DropOp = { op: "move"; folderId: number | null } | { op: "pin"; pinned: boolean };

/** What dropping `chat` on `target` changes (nothing: an empty list, and the
 * target does not accept the drop). The target decides the placement, so the
 * chat always shows up where it was dropped: a folder or Chats also unpins it,
 * Pinned pins it and leaves its folder alone. */
export function dropOps(chat: Conversation, target: DropTarget): DropOp[] {
  const folderId = chat.folderId ?? null;
  const ops: DropOp[] = [];
  switch (target.kind) {
    case "folder":
      if (folderId !== target.folderId) ops.push({ op: "move", folderId: target.folderId });
      if (chat.pinned) ops.push({ op: "pin", pinned: false });
      break;
    case "unfiled":
      if (folderId !== null) ops.push({ op: "move", folderId: null });
      if (chat.pinned) ops.push({ op: "pin", pinned: false });
      break;
    case "pinned":
      if (!chat.pinned) ops.push({ op: "pin", pinned: true });
      break;
  }
  return ops;
}
```

```ts
const dragging = ref(false);

function onDragStart(event: DragEvent): void {
  if (!props.draggable) {
    event.preventDefault();
    return;
  }
  event.dataTransfer?.setData("text/plain", props.chat.id);
  if (event.dataTransfer) event.dataTransfer.effectAllowed = "move";

}

function onDragEnd(): void {
  dragging.value = false;
  emit("dragEnd");
}
```

```ts
const over = ref(false);

function onDragOver(event: DragEvent): void {
  if (!props.accepting) return;
  event.preventDefault(); // without this the browser refuses the drop
  if (event.dataTransfer) event.dataTransfer.dropEffect = "move";
  over.value = true;
}

function onDragLeave(event: DragEvent): void {
  // Moving between the zone's own children fires dragleave too; only leaving the zone counts.
  const next = event.relatedTarget as Node | null;
  if (next && (event.currentTarget as Node).contains(next)) return;
  over.value = false;
}

function onDrop(event: DragEvent): void {
  if (!props.accepting) return;
  event.preventDefault();
  over.value = false;
  emit("drop");
}
```

```ts
// The chat being dragged. Set one tick after dragstart: Chrome cancels a drag
// whose source DOM changes inside the dragstart handler, and showing the drop
// zones changes the DOM.
const draggingId = ref<string | null>(null);
const dragging = computed(() => props.conversations.find((c) => c.id === draggingId.value) ?? null);

const layout = computed(() => buildLayout(props.conversations, props.folders, draggingId.value !== null));

function touchOnly(): boolean {
  return typeof window.matchMedia === "function" && window.matchMedia("(hover: none)").matches;
}

/** A row may be dragged unless something else owns it right now. */
function canDrag(c: Conversation): boolean {
  return !selecting.value && renamingId.value !== c.id && !answering(c) && !touchOnly();
}

function startDrag(c: Conversation): void {
  menu.value = null;
  setTimeout(() => {
    draggingId.value = c.id;
  }, 0);
}

function endDrag(): void {
  draggingId.value = null;
}

function targetOf(section: { kind: "pinned" | "folder" | "unfiled"; folder: ChatFolder | null }): DropTarget | null {
  if (section.kind === "pinned") return { kind: "pinned" };
  if (section.kind === "unfiled") return { kind: "unfiled" };
  return section.folder ? { kind: "folder", folderId: section.folder.id } : null;
}

function accepts(section: { kind: "pinned" | "folder" | "unfiled"; folder: ChatFolder | null }): boolean {
  const chat = dragging.value;
  const target = targetOf(section);
  return chat !== null && target !== null && dropOps(chat, target).length > 0;
}

function dropOn(section: { kind: "pinned" | "folder" | "unfiled"; folder: ChatFolder | null }): void {
  const chat = dragging.value;
  const target = targetOf(section);
  draggingId.value = null;
  if (!chat || !target) return;
  for (const op of dropOps(chat, target)) {
    if (op.op === "move") emit("move", chat.id, op.folderId);
    else emit("pin", chat.id, op.pinned);
  }
}

const HINTS = { pinned: "Drop here to pin", unfiled: "Drop here to take it out of its folder" } as const;
function hintOf(kind: "pinned" | "folder" | "unfiled"): string | null {
  return kind === "folder" ? null : HINTS[kind];
}
```
